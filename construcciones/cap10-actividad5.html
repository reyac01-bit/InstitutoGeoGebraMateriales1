# Capítulo 10, actividad 5 - Del arrastre a la justificación (Varignon)
# Pegar línea por línea en la barra de entrada (GeoGebra Classic 6, español).
# Comandos ejecutados en GeoGebra Classic por la API el 10-nov-2026 (véase LEEME.txt).
# Lo que va en comentario (#) se configura a mano y no formó parte de esa prueba.

A = (0, 0)
B = (5, 0.5)
C = (6, 4)
D = (1, 3)
cuad = Polígono(A, B, C, D)
M_1 = PuntoMedio(A, B)
M_2 = PuntoMedio(B, C)
M_3 = PuntoMedio(C, D)
M_4 = PuntoMedio(D, A)
var = Polígono(M_1, M_2, M_3, M_4)

# Fases de la justificación, gobernadas por un entero
est_fase = 0
# Marcar est_fase como auxiliar.
dAC = Segmento(A, C)
dBD = Segmento(B, D)
# Condición para mostrar (Avanzado) de dAC:  est_fase >= 1
# Condición para mostrar (Avanzado) de dBD:  est_fase >= 2
esq1 = Esquina(1)
txtPista = Si(est_fase == 1, Texto("Mire los triángulos ABC y ACD. ¿Qué dice el teorema del segmento medio?", esq1 + (0.4, 0.6)), Si(est_fase == 2, Texto("Ahora los triángulos ABD y BCD. Combine.", esq1 + (0.4, 0.6))))
# Botón "Pista 1", Al hacer clic:   Valor(est_fase, 1)
# Botón "Pista 2", Al hacer clic:   Valor(est_fase, 2)
# Botón "Reiniciar", Al hacer clic:
#   Valor(est_fase, 0)
#   Valor(A, (0, 0))
#   Valor(B, (5, 0.5))
#   Valor(C, (6, 4))
#   Valor(D, (1, 3))

# Comprobación numérica opcional (auxiliar)
parVar = SonParalelas(Recta(M_1, M_2), Recta(M_4, M_3)) && SonParalelas(Recta(M_2, M_3), Recta(M_1, M_4))
# Color dinámico de var (relleno):
#   Rojo:  Si(parVar, 0.1, 0.8)
#   Verde: Si(parVar, 0.6, 0.2)
#   Azul:  0.2
# Con el cuadrilátero cruzado, Área devuelve el valor absoluto del área con signo
# (comprobado el 10-nov-2026): la razón 1/2 se mantiene respecto de ese valor.
