# Ejemplo 2 - Campo de direcciones y método de Euler
g(x, y) = x - y
campo = CampoDirecciones(g, 25, 0.8, -4, -4, 6, 6)
P0 = (-3, 2)
h = Deslizador(0.05, 1, 0.05, 1, 180, false, true, false, false)
n = Deslizador(1, 60, 1, 1, 180, false, true, false, false)
est_verSol = false
euler = ListaIteración((x(A) + h, y(A) + h g(x(A), y(A))), A, {P0}, n)
poli = Poligonal(euler)
eulerFino = ListaIteración((x(A) + h / 8, y(A) + (h / 8) g(x(A), y(A))), A, {P0}, 8 n)
sol = ResuelveEDO(g, P0)          # condición: est_verSol
Pfin = Elemento(euler, n + 1)
PfinFino = Elemento(eulerFino, 8 n + 1)
err = abs(y(Pfin) - y(PfinFino))
txtErr = Si(est_verSol, Texto("Desviación en el extremo: " + err, (-3.8, 5.4)))
ce1 = CasillaEntrada(g)
# Color dinámico de poli: Rojo Si(err<0.5, err/0.5, 1)  Verde Si(err<0.5, 1-err/0.5, 0)  Azul 0
# Botón Reiniciar: Valor(P0,(-3,2)) Valor(h,0.5) Valor(n,12) Valor(est_verSol,false) ZoomAcerca(-4,-4,6,6)
