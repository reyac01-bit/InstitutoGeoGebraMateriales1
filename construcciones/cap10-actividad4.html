# Capítulo 10, actividad 4 - Las diagonales deciden
# Pegar línea por línea en la barra de entrada (GeoGebra Classic 6, español).
# Comandos ejecutados en GeoGebra Classic por la API el 10-nov-2026 (véase LEEME.txt).
# Lo que va en comentario (#) se configura a mano y no formó parte de esa prueba.

# Diagonal AC fija (marcar A y C como "Objeto fijo" en Configuración > Básico)
A = (-3, 0)
C = (3, 0)
dAC = Segmento(A, C)

# Controles libres (deslizadores: mín, máx, incremento, velocidad, ancho,
# es ángulo, horizontal, animado, aleatorio)
sc = Deslizador(0.15, 0.85, 0.05, 1, 160, false, true, false, false)
tc = Deslizador(0.15, 0.85, 0.05, 1, 160, false, true, false, false)
thc = Deslizador(20°, 160°, 1°, 1, 160, true, true, false, false)
kc = Deslizador(0.4, 1.4, 0.05, 1, 160, false, true, false, false)
# Valores iniciales sugeridos: sc = 0.4, tc = 0.6, thc = 60°, kc = 0.8.
# Leyendas: sc "posición del cruce sobre AC", tc "posición del cruce sobre BD",
# thc "ángulo entre diagonales", kc "razón BD/AC".

# Casillas de control (booleanos libres)
bisA = false
bisB = false
perp = false
cong = false
mismaRazon = false
# Casillas vinculadas, con leyendas:
#   bisA       "BD corta a AC en su punto medio"
#   bisB       "AC corta a BD en su punto medio"
#   perp       "Diagonales perpendiculares"
#   cong       "Diagonales congruentes"
#   mismaRazon "Se cortan en la misma razón"

# Dependencias
s = Si(bisA, 0.5, sc)
t = Si(bisB, 0.5, Si(mismaRazon, s, tc))
ang = Si(perp, 90°, thc)
L = Si(cong, 1, kc) * Distancia(A, C)
P = A + s (C - A)
u = Vector((cos(ang), sin(ang)))
B = P - t L u
D = P + (1 - t) L u
dBD = Segmento(B, D)
cuad = Polígono(A, B, C, D)
# Ocultar u y P (o dejar P visible como punto de cruce).

# Clasificación (auxiliares; tolerancias explícitas)
esPar = abs(s - 0.5) < 0.001 && abs(t - 0.5) < 0.001
esPerp = abs(ang - 90°) < 0.001
esCong = abs(L - Distancia(A, C)) < 0.001
esTrap = abs(s - t) < 0.001
esRomb = esPar && esPerp
esRect = esPar && esCong
esCuad = esRomb && esCong
esCometa = esPerp && (abs(s - 0.5) < 0.001 || abs(t - 0.5) < 0.001)
esTrapIso = esTrap && esCong

est_comprobar = false
# Casilla de control vinculada a est_comprobar, leyenda "Comprobar".
esq1 = Esquina(1)
txtClas = Si(est_comprobar, Texto(Si(esCuad, "Cuadrado.", Si(esRect, "Rectángulo: paralelogramo con diagonales congruentes.", Si(esRomb, "Rombo: paralelogramo con diagonales perpendiculares.", Si(esPar, "Paralelogramo: las diagonales se bisecan.", Si(esCometa, "Cometa: una diagonal es mediatriz de la otra.", Si(esTrapIso, "Trapecio isósceles: AB paralelo a DC y diagonales congruentes.", Si(esTrap, "Trapecio: AB es paralelo a DC.", "Cuadrilátero convexo sin más propiedades que las que usted impuso."))))))), esq1 + (0.4, 0.6)))
