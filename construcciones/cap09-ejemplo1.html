Se pegan línea por línea en GeoGebra Classic 6 en español. Las líneas con # se configuran a mano.

# Ejemplo 1 - Cuadrática en forma canónica (líneas para la barra de entrada)
a = Deslizador(-3, 3, 0.1, 1, 180, false, true, false, false)
h = Deslizador(-5, 5, 0.5, 1, 180, false, true, false, false)
k = Deslizador(-5, 5, 0.5, 1, 180, false, true, false, false)
est_verificar = false          # convertir en casilla (Configuración) o: CasillaControl("Verificar")
est_intentos = 0               # marcar como auxiliar
f(x) = a (x - h)^2 + k
V = (h, k)
eje: x = h
raices = {Raíz(f)}
txtF = Texto("f(x) = " + FórmulaTexto(f), (-5.6, 7.2))
txtConcav = Texto(Si(a > 0, "Cóncava hacia arriba", Si(a < 0, "Cóncava hacia abajo", "a = 0: la parábola degenera en una recta")), (-5.6, 6.4))
txtV = Si(est_verificar, Texto("V = " + FórmulaTexto(V), (x(V) + 0.3, y(V) + 0.8)))
# Condición para mostrar (Configuración > Avanzado):
#   V, eje: est_verificar        raices: est_verificar && a != 0
# Color dinámico de f: Rojo Si(a<0,0.85,0.10)  Verde 0.25  Azul Si(a<0,0.10,0.75)
# Botón "Reiniciar", guion Al hacer clic:
#   Valor(a, 1)  Valor(h, 0)  Valor(k, 0)  Valor(est_verificar, false)
#   Valor(est_intentos, est_intentos + 1)  ZoomAcerca(-6, -4, 6, 8)
