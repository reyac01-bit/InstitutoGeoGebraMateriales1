# Ejemplo 6 - La derivada en cuatro fases
est_fase = 1                      # auxiliar
f(x) = x^2 / 2 + 1
a = Deslizador(-2, 3, 0.1, 1, 180, false, true, false, false)
hh = Deslizador(0.01, 3, 0.01, 1, 180, false, true, false, false)
P = (a, f(a))
Q = (a + hh, f(a + hh))
sec = Recta(P, Q)                                  # condición: est_fase >= 1
m_sec = (f(a + hh) - f(a)) / hh
triangulo = Polígono(P, (a + hh, f(a)), Q)         # condición: est_fase >= 2
esq4 = Esquina(4)
txtCoc = Texto("m = " + m_sec, esq4 + (0.4, -2.2))  # condición: est_fase >= 2
tang = Recta(P, Vector((1, f'(a))))                # condición: est_fase >= 3
R = (a, f'(a))                                     # condición: est_fase >= 3; activar Rastro
df(x) = f'(x)                                      # condición: est_fase >= 4
txtFase = Texto("Fase " + est_fase + " de 4", esq4 + (0.4, -0.5))
