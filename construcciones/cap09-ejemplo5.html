# Ejemplo 5 - Regresión lineal con residuos condicionados
datos = {(1, 2.1), (2, 2.9), (3, 3.6), (4, 4.8), (5, 5.1), (6, 6.3), (7, 6.8), (8, 8.1), (9, 8.7), (10, 9.9)}
Q = (8, 9)
est_incluirQ = true               # casilla "Incluir Q en el ajuste"
est_verCuadrados = false          # casilla "Mostrar cuadrados"
conjunto = Si(est_incluirQ, Añade(datos, Q), datos)
recta = AjusteLineal(conjunto)
fr = AjustePolinómico(conjunto, 1)
r2 = RCuadrado(conjunto, fr)
resPos = Zip(Si(y(W) > fr(x(W)), Segmento(W, (x(W), fr(x(W))))), W, conjunto)
resNeg = Zip(Si(y(W) <= fr(x(W)), Segmento(W, (x(W), fr(x(W))))), W, conjunto)
cuadrados = Zip(Polígono((x(W), y(W)), (x(W), fr(x(W))), (x(W) + (y(W) - fr(x(W))), fr(x(W))), (x(W) + (y(W) - fr(x(W))), y(W))), W, conjunto)
SCE = SumaErroresCuadrados(conjunto, fr)
esq4 = Esquina(4)
txtAjuste = Texto("y = " + FórmulaTexto(fr) + "\quad R^2 = " + r2 + "\quad SCE = " + SCE, esq4 + (0.4, -1.4))
# Condición para mostrar de cuadrados: est_verCuadrados
