Se pegan línea por línea en GeoGebra Classic 6 en español. Las líneas con # se configuran a mano.

# Capítulo 10, actividad 2 - Un paralelogramo de tres maneras
# Pegar línea por línea en la barra de entrada (GeoGebra Classic 6, español).
# Comandos ejecutados en GeoGebra Classic por la API el 10-nov-2026 (véase LEEME.txt).
# Lo que va en comentario (#) se configura a mano y no formó parte de esa prueba.

# 1. Por paralelas
A = (0, 0)
B = (4, 0)
C = (5, 2.5)
r1 = Recta(C, Recta(A, B))
r2 = Recta(A, Recta(B, C))
D = Interseca(r1, r2)
p1 = Polígono(A, B, C, D)
# Ocultar r1 y r2 o marcarlas auxiliares.

# 2. Por traslación
E = (7, 0)
F = (11, 0)
G = (12, 2.5)
H = Traslada(G, Vector(F, E))
p2 = Polígono(E, F, G, H)

# 3. Por diagonales que se bisecan
I = (0, -5)
K = (5, -3)
M = PuntoMedio(I, K)
J = (1, -6.5)
L = M + Vector(J, M)
p3 = Polígono(I, J, K, L)

# Comprobaciones numéricas (auxiliares; visibles con est_medir)
est_medir = false
# Casilla de control vinculada a est_medir, leyenda "Mostrar comprobaciones".
par1 = SonParalelas(Recta(A, B), Recta(D, C))
par2 = SonParalelas(Recta(E, H), Recta(F, G))
par3 = SonParalelas(Recta(I, J), Recta(L, K))
ladoAB = Distancia(A, B)
ladoDC = Distancia(D, C)
# Corregido el 10-nov-2026: con (D, A, B) y (B, C, D) GeoGebra devolvía el ángulo reflejo (291.8°).
angA = Ángulo(B, A, D)
angC = Ángulo(D, C, B)
# Condición para mostrar (Avanzado) de par1, par2, par3, ladoAB, ladoDC, angA, angC:  est_medir
# Las rectas creadas dentro de SonParalelas(...) no se dibujan; si GeoGebra las crea
# como objetos auxiliares visibles, ocultarlas.
