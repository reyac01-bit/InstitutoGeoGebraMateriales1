# Ejemplo 3 - Ángulo inscrito con autoevaluación
O = (0, 0)
r = Deslizador(1, 4, 0.5, 1, 180, false, true, false, false)
c = Circunferencia(O, r)
A = Punto(c)
B = Punto(c)
C = Punto(c)
alfa = Ángulo(A, C, B)
central = Ángulo(A, O, B)
est_intentos = 0
est_correcto = false
est_pista = false
respuesta = 2
esq1 = Esquina(1)
esq4 = Esquina(4)
ce1 = CasillaEntrada(respuesta)
txtRes = Texto(Si(est_correcto, "Correcto. Mueva C y compruebe de nuevo.", Si(est_intentos == 0, "Escriba su conjetura.", "Todavía no. Observe qué ocurre al mover C.")), esq4 + (0.4, -0.5))
txtPista = Si(est_pista, Texto("Pista: compare las medidas cuando C está lejos de A y de B.", esq4 + (0.4, -1.2)))
# alfa: en Configuración > Básico, marcar "Ángulo entre 0° y 180°" (mide el ángulo inscrito geométrico)
# central: dejar "Ángulo entre 0° y 360°" (mide el arco de A a B en sentido antihorario)

enArco = gammaC < central
